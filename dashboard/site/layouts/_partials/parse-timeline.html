"error" -}} {{- range $index, $raw := split .source "\n" -}} {{- $line := strings.TrimSuffix "\r" $raw -}} {{- $task := findRESubmatch `^[-*+] \[([xX /!])\][ \t]*(.*)$` $line -}} {{- if $task -}} {{- if $current -}}{{- $steps = $steps | append $current -}}{{- end -}} {{- $match := index $task 0 -}} {{- $text := index $match 2 -}} {{- $open := gt (len (findRE ` \{open\}$` $text)) 0 -}} {{- $text = replaceRE ` \{open\}$` "" $text -}} {{- $current = dict "text" $text "status" (index $statuses (index $match 1)) "content" "" "open" $open -}} {{- else if and $current (or (findRE `^( {2}|\t)` $line) (not (strings.TrimSpace $line))) -}} {{- $body := replaceRE `^( {2}|\t)` "" $line -}} {{- if $current.content -}}{{- $body = printf "%s\n%s" $current.content $body -}}{{- end -}} {{- $current = merge $current (dict "content" $body) -}} {{- else if not (strings.TrimSpace $line) -}} {{- else if and (not $current) (strings.HasPrefix $line "# ") -}} {{- $model = merge $model (dict "title" (strings.TrimSpace (strings.TrimPrefix "# " $line))) -}} {{- else if and (not $current) (findRE `^collapse: [0-9]+$` $line) -}} {{- $model = merge $model (dict "collapse" (int (strings.TrimPrefix "collapse: " $line))) -}} {{- else -}} {{- errorf "timeline: line %d needs a task (- [ ] Title) or two-space-indented details at %s" (add $index 1) $.Position -}} {{- end -}} {{- end -}} {{- if $current -}}{{- $steps = $steps | append $current -}}{{- end -}} {{- return (merge (dict "steps" $steps) $model) -}}
